        function handleDriveImgError(img, fileId, fallbackLabel) {
            const step = parseInt(img.getAttribute('data-retry') || '0', 10);
            if (step === 0) {
                img.setAttribute('data-retry', '1');
                img.src = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=w1000';
            } else if (step === 1) {
                img.setAttribute('data-retry', '2');
                img.src = 'https://drive.google.com/uc?export=view&id=' + fileId;
            } else {
                img.onerror = null;
                const text = encodeURIComponent(fallbackLabel || 'Phu Lieu Toc Loan Nhung');
                img.src = 'https://placehold.co/600x600/fdf2f2/b81d24?text=' + text;
            }
        }

            const sections = document.querySelectorAll('[data-cat-section]');
            const bar = document.getElementById('filter-result-bar');
            const label = document.getElementById('activeFilterName');

            if (catKey === 'ALL') {
                bar.classList.add('hidden');
                sections.forEach(s => s.classList.remove('hidden'));
                return;
            }

            bar.classList.remove('hidden');
            label.textContent = catKey;

            sections.forEach(sec => {
                const secCat = sec.getAttribute('data-cat-section');
                if (
                    secCat === catKey ||
                    (catKey === 'DỤNG CỤ NỐI MI' && secCat === 'SẢN PHẨM LÀM NAIL') ||
                    ((catKey === 'CHĂM SÓC DA & BODY' || catKey === 'MỸ PHẨM MAKE UP') && secCat === 'GHẾ TÓC, NAIL, SPA')
                ) {
                    sec.classList.remove('hidden');
                } else {
                    sec.classList.add('hidden');
                }
            });
        }

        // Tìm kiếm sản phẩm
        function executeSearch(keyword) {
            const q = (keyword || '').trim().toLowerCase();
            if (!q) {
                renderAllProductSections();
                filterByCategory('ALL');
                return;
            }
            const matched = PRODUCTS.filter(p =>
                p.name.toLowerCase().includes(q) || p.category.toLowerCase().includes(q)
            );
            const firstGrid = document.getElementById('grid-ban-nhieu');
            if (firstGrid) {
                firstGrid.innerHTML = matched.length
                    ? matched.map(createProductCardHTML).join('')
                    : `<div class="col-span-full bg-white p-6 rounded-xl text-center border border-gray-200 text-xs sm:text-sm text-gray-600">
                         Không tìm thấy sản phẩm khớp với từ khóa "<strong>${keyword}</strong>". Quý khách vui lòng liên hệ Zalo <strong>0397.435.668</strong> để kiểm tra kho nhanh nhất!
                       </div>`;
                document.getElementById('sec-ban-nhieu').scrollIntoView({ behavior: 'smooth' });
            }
        }

        // Giỏ hàng
        function addToCart(productId) {
            const product = PRODUCTS.find(p => p.id === productId);
            if (!product) return;
            const existing = cart.find(item => item.id === productId);
            if (existing) {
                existing.qty += 1;
            } else {
                cart.push({ ...product, qty: 1 });
            }
            updateCartUI();
            openCartModal();
        }

        function changeCartQty(productId, delta) {
            const idx = cart.findIndex(i => i.id === productId);
            if (idx === -1) return;
            cart[idx].qty += delta;
            if (cart[idx].qty <= 0) cart.splice(idx, 1);
            updateCartUI();
        }

        function updateCartUI() {
            const count = cart.reduce((sum, i) => sum + i.qty, 0);
            const total = cart.reduce((sum, i) => sum + i.price * i.qty, 0);
            document.getElementById('cartCountBadge').textContent = count;
            document.getElementById('cartTotalAmount').textContent = formatVND(total);

            const container = document.getElementById('cartItemsContainer');
            if (!cart.length) {
                container.innerHTML = `<p class="text-center text-xs sm:text-sm text-gray-500 py-6">Giỏ hàng của bạn đang trống. Hãy chọn sản phẩm để đặt mua hoặc nhận báo giá sỉ!</p>`;
                return;
            }
            container.innerHTML = `<div class="divide-y divide-gray-100">` + cart.map(item => `
                <div class="py-3 flex items-center justify-between gap-3">
                    <div class="min-w-0 flex-1">
                        <p class="text-xs sm:text-sm font-semibold text-gray-800">${item.name}</p>
                        <p class="text-xs text-primary font-bold mt-0.5">${formatVND(item.price)}</p>
                    </div>
                    <div class="flex items-center gap-2 shrink-0">
                        <button type="button" onclick="changeCartQty('${item.id}', -1)" class="w-7 h-7 rounded bg-gray-100 hover:bg-gray-200 font-bold text-xs">-</button>
                        <span class="text-xs font-bold w-5 text-center">${item.qty}</span>
                        <button type="button" onclick="changeCartQty('${item.id}', 1)" class="w-7 h-7 rounded bg-gray-100 hover:bg-gray-200 font-bold text-xs">+</button>
                    </div>
                </div>
            `).join('') + `</div>`;
        }

        function openCartModal() {
            updateCartUI();
            const modal = document.getElementById('cartModal');
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeCartModal() {
            const modal = document.getElementById('cartModal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        // Điều khiển 2 Banner Tự Trượt Chuyên Nghiệp
        let currentBanner = 0;
        let bannerInterval = null;

        function goToBannerSlide(index) {
            currentBanner = (index + 2) % 2;
            const slide0 = document.getElementById('banner-slide-0');
            const slide1 = document.getElementById('banner-slide-1');
            const dot0 = document.getElementById('banner-dot-0');
            const dot1 = document.getElementById('banner-dot-1');

            if (currentBanner === 0) {
                slide0.classList.remove('opacity-0', 'invisible', 'z-0');
                slide0.classList.add('opacity-100', 'visible', 'z-10');
                slide1.classList.remove('opacity-100', 'visible', 'z-10');
                slide1.classList.add('opacity-0', 'invisible', 'z-0');
                dot0.className = 'w-6 h-2 rounded-full bg-gold transition-all';
                dot1.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
            } else {
                slide1.classList.remove('opacity-0', 'invisible', 'z-0');
                slide1.classList.add('opacity-100', 'visible', 'z-10');
                slide0.classList.remove('opacity-100', 'visible', 'z-10');
                slide0.classList.add('opacity-0', 'invisible', 'z-0');
                dot1.className = 'w-6 h-2 rounded-full bg-gold transition-all';
                dot0.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
            }
        }

        function changeBannerSlide(step) {
            goToBannerSlide(currentBanner + step);
            resetBannerAutoSlide();
        }

        function resetBannerAutoSlide() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                goToBannerSlide(currentBanner + 1);
            }, 4500);
        }
